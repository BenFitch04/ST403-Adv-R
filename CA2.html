# pdf of Kumaraswamy
dkumaraswamy <- function(x, a, b) {
  result <- a* b * x^(a - 1)* (1 - x^a)^(b - 1)
  return(result)
}

dkumaraswamy(x =c(0.3, 0.6, 0.9), a = 1, b = 2)
## [1] 1.4 0.8 0.2
#default arguments
dkumaraswamy2 <- function(x, a = 1, b = 1) {
  #check if a and b valid
  if (a <= 0 || b <= 0) {
    stop("a and b must be greater than zero")}

  # if both are 1 it has unif distrib
  if (a == 1 & b == 1) {
    return(dunif(x))}
  
  result <- a* b * x^(a - 1)* (1 - x^a)^(b - 1)
  return(result)
}

rkumaraswamy <- function(a, b, n) {
  u <- runif(n) #get n unif nums
  #inverse CDF
  x <- (1 -(1 - u)^(1/b))^(1/a)
  return(x)
}

M <- 2^32

n <- 10000

x <- numeric(n)

x[1] <- 2345 #initial seed

#generate sequence
for (i in 2:n) {
  x[i] <- (a * x[i-1] + b) %% M #Marsaglia formula

#change to unif rand nums
u <- x / M

U <- data.frame(u1 = u[1:(n-2)],
                u2 = u[2:(n-1)],
                u3 = u[3:n])

#special ordering ui−1 < ui+1 < ui
event <- U$u1 < U$u3 & U$u3 < U$u2

#percentage event occurs (should be 1/6)

box_muller <- function(numpairs) {
  #unif rand nums
  u1 <- runif(numpairs)
  u2 <- runif(numpairs)
  
  theta <- 2*pi * u1
  R <- sqrt(2 * -log(u2))

  #make standard norm variables
  x1 <- R * cos(theta)
  x2 <- R * sin(theta)

  #return both vectors 
  return(c(x1, x2))}
